@csrf_exempt
def create_checkout_session(request):
    if request.method == 'GET':
        lst=[]
        order, created = Order.objects.get_or_create(customer=request.user.customer,complete=False)
        orderitems = order.orderitem_set.all()
        for itm in orderitems:
                lst.append({
                    'price_data': {
                        'currency': 'usd',
                        'product_data': {
                            'name': itm.product.title,
                        },
                        'unit_amount': int(float(itm.product.price) * 100),  # Amount in cents ($400 ke liye * 100 zaroori hai)
                    },
                    'quantity': itm.quantity,
                }, )
        print(lst)
        domain_url = 'http://127.0.0.1:8000/'
        stripe.api_key = settings.STRIPE_SECRET_KEY
        try:
            checkout_session = stripe.checkout.Session.create(
                success_url=domain_url + 'success?session_id={CHECKOUT_SESSION_ID}',
                cancel_url=domain_url + 'cancelled/',
                # payment_method_types=['card'],  <-- Yeh line hata di gayi hai
                mode='payment',
                line_items=lst,
                metadata={
                    'cust': request.user.customer.id,
                    'adr': request.GET.get("query")
                }
            )
            return JsonResponse({'sessionId': checkout_session['id']})
        except Exception as e:
            return JsonResponse({'error': str(e)})

def register(request):
    if request.method=="POST":
        user_form=UserForm(request.POST)
        cust_form=CustomerForm(request.POST)
        if user_form.is_valid() and cust_form.is_valid():
            user=user_form.save()
            cust=cust_form.save(commit=False)
            cust.user=user
            cust.save()
            login(request, user)
        return redirect("index")
    else:
        frm=UserForm()
        cstfrm=CustomerForm()
        return render(request,"registration/register.html",
                      {"usrfrm":frm,"cstfrm":cstfrm})

def updatecart(request):
    if request.user.is_authenticated:
        if request.method=="POST":
            if request.POST.get("btnupd"):
                pids = [(int)(ctrl[3:]) for ctrl in request.POST if 'txt' in ctrl]
                customer = request.user.customer
                order, created = Order.objects.get_or_create(customer=customer, complete=False)
                prd = Product.objects.get(id=pids[0])
                orditm, crt = OrderItem.objects.get_or_create(order=order, product=prd)
                qty=(int)(request.POST.get("txt"+str(pids[0])))
                if qty<=0:
                    orditm.delete()
                else:
                    orditm.quantity=qty
                    orditm.save()

            elif request.POST.get("btndel"):
                pids=[(int)(ctrl[3:]) for ctrl in request.POST if 'txt' in ctrl]
                customer = request.user.customer
                order, created = Order.objects.get_or_create(customer=customer, complete=False)
                prd = Product.objects.get(id=pids[0])
                orditm,crt=OrderItem.objects.get_or_create(order=order,product=prd)
                orditm.delete()
            else:
                pids=[(int)(ctrl[3:]) for ctrl in request.POST if 'btn' in ctrl]
                customer=request.user.customer
                order,created=Order.objects.get_or_create(customer=customer,complete=False)
                prd=Product.objects.get(id=pids[0])
                orderitem,created=OrderItem.objects.get_or_create(order=order,product=prd)
                orderitem.quantity+=1
                orderitem.save()
        return redirect("cart")
    else:
        return redirect("login")

@csrf_exempt
def stripe_webhook(request):
    stripe.api_key = settings.STRIPE_SECRET_KEY
    endpoint_secret = settings.STRIPE_ENDPOINT_SECRET
    payload = request.body
    sig_header = request.META.get('HTTP_STRIPE_SIGNATURE')
    event = None

    try:
        event = stripe.Webhook.construct_event(
            payload, sig_header, endpoint_secret
        )
    except ValueError as e:
        return HttpResponse(status=400)
    except stripe.error.SignatureVerificationError as e:
        return HttpResponse(status=400)

    # Handle the checkout.session.completed event
    if event['type'] == 'checkout.session.completed':
        print("Payment was successful.")

        # YAHAN BADLAAV HAI: Session ko dictionary mein convert kiya gaya hai
        session = event["data"]["object"].to_dict()

        metadata = session.get("metadata", {})

        cust_id = metadata.get("cust")
        addr_string = metadata.get("adr")

        print(f"DEBUG DATA -> Customer ID: {cust_id}, Address: {addr_string}")

        if cust_id and addr_string:
            try:
                # 1. Pehle Customer object retrieve karein
                customer_obj = Customer.objects.get(id=cust_id)

                # 2. Sahi customer instance ke sath order get/create karein
                ord, created = Order.objects.get_or_create(customer=customer_obj, complete=False)
                ord.transaction_id = session.get("payment_intent")
                ord.complete = True
                ord.save()

                # 3. Address split karke ShippingAddress create karein
                adr, cty, sta, zipcode = addr_string.split("|")

                ShippingAddress.objects.create(
                    customer=customer_obj,
                    order=ord,
                    address=adr,
                    city=cty,
                    state=sta,
                    zipcode=zipcode
                )
                print("Shipping address saved successfully!")

            except Customer.DoesNotExist:
                print("Error: Customer not found in database.")
            except Exception as e:
                print(f"Error saving order/address: {e}")
        else:
            # Agar metadata se data nahi mila toh yahan print ho jayega
            print(f"WARNING: Metadata missing! cust_id: {cust_id}, addr_string: {addr_string}")

    return HttpResponse(status=200)
